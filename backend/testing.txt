  Step 5 — Check PostgreSQL directly

  -- Check the content item was created and its status
SELECT id, title, status, content_type, created_at 
FROM content_items 
ORDER BY created_at DESC LIMIT 5;

-- Check the job status and progress
SELECT id, status, progress, progress_message, error_message, started_at, completed_at 
FROM processing_jobs 
ORDER BY created_at DESC LIMIT 5;

-- Check generated outputs (summary, flashcards, quiz, etc.)
SELECT output_type, language, status, prompt_tokens, completion_tokens, created_at 
FROM ai_outputs 
ORDER BY created_at DESC LIMIT 10;

-- Check what chunks were stored (count per content item)
SELECT content_item_id, COUNT(*) as chunk_count 
FROM content_chunks 
GROUP BY content_item_id;
